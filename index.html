function doGet(e) { const code = e && e.parameter ? e.parameter.code : ""; const callback = e && e.parameter ? e.parameter.callback : ""; let result; if (code) { result = checkIn(code); } else { result = { success: false, error: "Brak kodu QR" }; } const json = JSON.stringify(result); // Obsługa JSONP dla skanera na GitHub Pages if (callback) { return ContentService .createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function checkIn(code) { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName("UCZESTNICY"); if (!sheet) { return { success: false, error: "Nie znaleziono arkusza UCZESTNICY" }; } const lock = LockService.getScriptLock(); try { lock.waitLock(10000); const data = sheet.getDataRange().getValues(); code = String(code).trim().toLowerCase(); for (let i = 1; i < data.length; i++) { // KOLUMNA B — KOD BILETU const ticketCode = String(data[i][1]) .trim() .toLowerCase(); if (ticketCode === code) { // KOLUMNA D — IMIĘ I NAZWISKO const name = data[i][3]; // KOLUMNA E — PAKIET const pack = String(data[i][4]).trim(); // KOLUMNA F — STATUS WEJŚCIA const status = String(data[i][5]).trim(); // KOLUMNA G — POSIŁEK const meals = data[i][6]; // KOLOR OPASKI let bandColor = ""; const packLower = pack.toLowerCase(); if (packLower.includes("jednodniowy")) { bandColor = "CZERWONA"; } else if (packLower.includes("silver")) { bandColor = "ZIELONA"; } else if (packLower.includes("gold")) { bandColor = "ZŁOTA"; } else if (packLower.includes("vip")) { bandColor = "NIEBIESKA"; } else if (packLower.includes("bankiet")) { bandColor = "CZARNA"; } else if (packLower.includes("platinum")) { bandColor = "RÓŻOWA"; } // ========================================== // BILET JUŻ ZESKANOWANY // ========================================== if (status === "WESZŁA") { const oldTime = data[i][7]; let formattedOldTime = ""; if (oldTime instanceof Date) { formattedOldTime = Utilities.formatDate( oldTime, Session.getScriptTimeZone(), "HH:mm" ); } else { formattedOldTime = String(oldTime || ""); } return { success: false, already: true, name: name, pack: pack, meals: meals, bandColor: bandColor, entryTime: formattedOldTime }; } // ========================================== // NOWE WEJŚCIE // ========================================== const now = new Date(); // KOLUMNA F — STATUS sheet .getRange(i + 1, 6) .setValue("WESZŁA"); // KOLUMNA H — CZAS sheet .getRange(i + 1, 8) .setValue(now); // KOLUMNA I — KOLOR OPASKI sheet .getRange(i + 1, 9) .setValue(bandColor); return { success: true, name: name, pack: pack, meals: meals, bandColor: bandColor, entryTime: Utilities.formatDate( now, Session.getScriptTimeZone(), "HH:mm" ) }; } } // ========================================== // NIE ZNALEZIONO BILETU // ========================================== return { success: false, notFound: true }; } finally { lock.releaseLock(); } }
